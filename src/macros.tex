% These macros make it easy to include content only in specific variants
% Usage: \ifembedded{content}, \ifsoftware{content}, etc.

% Helper macro to check if TARGET matches a specific value
\makeatletter
\newcommand{\@checkTarget}[3]{%
  \ifx\TARGET\@undefined%
    #3% If TARGET is undefined, use third argument (usually empty)
  \else%
    \edef\@tempa{\TARGET}%
    \edef\@tempb{#1}%
    \ifx\@tempa\@tempb%
      #2% Match found, use second argument
    \else%
      #3% No match, use third argument
    \fi%
  \fi%
}

% Main conditional macros - easy to use and read
\newcommand{\ifembedded}[1]{\@checkTarget{embedded}{#1}{}}
\newcommand{\ifsoftware}[1]{\@checkTarget{software}{#1}{}}
\newcommand{\ifhardware}[1]{\@checkTarget{hardware}{#1}{}}
\newcommand{\iffirmware}[1]{\@checkTarget{firmware}{#1}{}}
